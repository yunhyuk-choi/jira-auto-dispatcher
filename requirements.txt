# jira-auto-dispatcher — 최소 런타임 의존성
# 실제 버전 핀은 Phase 1(config/state)·Phase 6(도커) 단계에서 확정한다.
flask          # central 관리 UI/HTTP, worker 최소 헬스
pyyaml         # config 로드
requests       # central Jira REST + worker→central HTTP 폴링
APScheduler    # central 재개/야간 드레인 스케줄러
docker         # central spawner — 사용자 worker 컨테이너 동적 기동(Docker SDK)
